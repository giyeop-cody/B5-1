SQL
-- [Q12][서브쿼리] 최고 가격과 같은 모든 메뉴가 포함된 주문을 조회한다.
SELECT *
FROM orders
WHERE menu_id IN (
    SELECT id
    FROM menus
    WHERE price = (SELECT MAX(price) FROM menus)
)
ORDER BY id;
