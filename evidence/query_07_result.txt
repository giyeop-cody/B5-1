=== Q07 [LEFT JOIN] 전체 샘플 기간에 주문 이력이 한 번도 없는 메뉴를 찾는다. ===

SQL
-- [Q07][LEFT JOIN] 전체 샘플 기간에 주문 이력이 한 번도 없는 메뉴를 찾는다.
SELECT m.id, m.name, m.price
FROM menus m
LEFT JOIN orders o ON m.id = o.menu_id
WHERE o.id IS NULL
ORDER BY m.id;
