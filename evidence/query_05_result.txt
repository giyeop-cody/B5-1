-- [조인 1 - INNER] 주문 내역에 대해 테이블 번호와 주문된 메뉴명, 수량, 주문 상태를 결합하여 실시간 주방 모니터 조회
SELECT t.table_number, m.name AS menu_name, o.quantity, o.status, o.order_time FROM orders o INNER JOIN store_tables t ON o.table_id = t.id INNER JOIN menus m ON o.menu_id = m.id WHERE o.status != 'CANCELLED';
