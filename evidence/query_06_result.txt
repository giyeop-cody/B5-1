=== Q06 [INNER JOIN] 주류 카테고리 3종의 카테고리명·메뉴명·가격을 조회한다. ===

SQL
-- [Q06][INNER JOIN] 주류 카테고리 3종의 카테고리명·메뉴명·가격을 조회한다.
SELECT c.name AS category_name, m.name AS menu_name, m.price
FROM menus m
INNER JOIN menu_categories c ON m.category_id = c.id
WHERE c.name IN ('소주/맥주', '하이볼/칵테일', '프리미엄 전통주')
ORDER BY c.id, m.id;

RESULT
